      // DONE: [RESOLVED] Connected conductor endpoint POST /api/omni/acp/agents/dispatch
      const response = await fetch('/api/ide/exec', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          command: `acp dispatch --agent=${payload.agentId} --file=${payload.context.activeFilePath} "${payload.prompt}"`
        })
      });
      const resData = await response.json();

      // STAGE MOCK/GENERIC SHADOW DIFF FOR REVIEV
      setStagedDiff({
        path: activePath,
        originalCode: currentTab.content,
        stagedCode: currentTab.content + `\n// Generated by ${selectedAgent} for: ${composerPrompt}\n`,
        prompt: composerPrompt,
        agentId: selectedAgent,
        timestamp: Date.now()
      });
      setComposerOpen(false);
    } catch (err) {
      console.error('[Forge IDE] Composer dispatch failure:', err);
    }
  };

  const acceptDiff = async () => {
    if (!stagedDiff) return;
    await saveFile(stagedDiff.path, stagedDiff.stagedCode);
    setStagedDiff(null);
  };

  const rejectDiff = () => {
    setStagedDiff(null);
  };

  // --- TERMINAL EXECUTION ---
  const executeCommand = async () => {
    if (!currentCommand.trim()) return;
    const cmd = currentCommand;
    setCurrentCommand('');
    setIsExecuting(true);

    const logEntry: TerminalExecutionLog = {
      id: Math.random().toString(36).substring(7),
      command: cmd,
      timestamp: new Date().toLocaleTimeString(),
      output: 'Executing...\n',
      status: 'running'
    };

    setExecHistory((prev) => [...prev, logEntry]);

    try {
      const res = await fetch('/api/ide/exec', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ command: cmd })
      });
      const data = await res.json();
      setExecHistory((prev) =>
        prev.map((item) =>
          item.id === logEntry.id
            ? { ...item, output: data.output || data.stdout || 'Executed successfully.', status: 'completed' }
            : item
        )
      );
    } catch (err: any) {
      setExecHistory((prev) =>
        prev.map((item) =>
          item.id === logEntry.id
            ? { ...item, output: `Error: ${err.message}`, status: 'error' }
            : item
        )
      );
    } finally {
      setIsExecuting(false);
    }
  };

  // --- MONACO SETUP ---
  const handleEditorMount = (editor: any, monaco: Monaco) => {
    editorRef.current = editor;

    // KEYBINDING: Ctrl+K or Cmd+K -> Open Inline Composer
    editor.addCommand(monaco.KeyMod.CtrlCmd | monaco.KeyCode.KeyK, () => {
      const position = editor.getScrolledVisiblePosition(editor.getPosition());
      if (position) {
        setComposerPosition({ top: position.top + 40, left: position.left + 50 });
      }
      setComposerOpen(true);
    });
  };

  const currentTab = activeTabs.find((t) => t.path === activePath);

  return (
    <div className="relative flex flex-col h-full w-full bg-slate-950 text-slate-100 font-mono overflow-hidden">
      
      {/* 1. AMBIENT CANVAS BACKDROP */}
      <div 
        className="absolute inset-0 pointer-events-none z-0 transition-opacity duration-300"
        style={{ opacity: canvasOpacity }}
      >
        <iframe 
          src="/artifacts/sovereign_canvas.html" 
          className="w-full h-full border-0"
          title="Sovereign Ambient Backdrop"
        />
      </div>

      {/* 2. TOP TOOLBAR & WORKER DELEGATION BAR */}
      <header className="relative z-10 flex items-center justify-between px-4 py-2 border-b border-slate-800 bg-slate-900/80 backdrop-blur-md">
        <div className="flex items-center gap-3">
          <span className="text-amber-400 font-bold tracking-wider text-sm flex items-center gap-1.5">
            <Sparkles className="w-4 h-4" /> FORGE PRO IDE
          </span>
          <span className="text-xs text-slate-500">|</span>
          <div className="flex items-center gap-1 text-xs">
            <Sliders className="w-3.5 h-3.5 text-slate-400" />
            <span className="text-slate-400">Canvas Opacity:</span>
            <input
              type="range"
              min="0"
              max="0.8"
              step="0.05"
              value={canvasOpacity}
              onChange={(e) => setCanvasOpacity(parseFloat(e.target.value))}
              className="w-20 accent-amber-400 h-1 bg-slate-700 rounded-lg cursor-pointer"
            />
          </div>
        </div>

        {/* WORKER DELEGATION SELECTOR */}
        <div className="flex items-center gap-2 bg-slate-950/80 px-3 py-1 rounded-md border border-slate-800">
          <Layers className="w-4 h-4 text-emerald-400" />
          <span className="text-xs text-slate-400 font-medium">Worker:</span>
          {(['auto', 'codex', 'claude', 'aider'] as const).map((agent) => (
            <button
              key={agent}
              onClick={() => setSelectedAgent(agent)}
              className={`px-2 py-0.5 text-xs rounded transition-colors ${
                selectedAgent === agent
                  ? 'bg-amber-500/20 text-amber-300 font-bold border border-amber-500/40'
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800'
              }`}
            >
              {agent.toUpperCase()}
            </button>
          ))}
        </div>
      </header>

      {/* 3. MAIN WORKSPACE ROW */}
      <div className="relative z-10 flex flex-1 overflow-hidden">
        
        {/* FILE TREE SIDEBAR */}
        <aside className="w-64 border-r border-slate-800 bg-slate-900/60 backdrop-blur-sm flex flex-col">
          <div className="flex items-center justify-between px-3 py-2 border-b border-slate-800/60">
            <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider">Explorer</span>
            <div className="flex items-center gap-1">
              <button 
                onClick={() => createFile('')}
                className="p-1 hover:bg-slate-800 rounded text-slate-400 hover:text-slate-200"
                title="New File"
              >
                <Plus className="w-3.5 h-3.5" />
              </button>
              <button 
                onClick={() => fetchTree()}
                className="p-1 hover:bg-slate-800 rounded text-slate-400 hover:text-slate-200"
                title="Refresh Tree"
              >
                <RefreshCw className="w-3.5 h-3.5" />
              </button>
            </div>
          </div>
          <div className="flex-1 overflow-y-auto p-2 text-xs">
            {renderTree(tree, openFile, deleteFile, activePath, activeTabs)}
          </div>
        </aside>

        {/* EDITOR AREA */}
        <main className="flex-1 flex flex-col bg-slate-950/40 relative overflow-hidden">
          
          {/* TAB BAR */}
          <div className="flex items-center bg-slate-900/40 border-b border-slate-800 overflow-x-auto">
            {activeTabs.map((tab) => (
              <div
                key={tab.path}
                onClick={() => setActivePath(tab.path)}
                className={`flex items-center gap-2 px-3 py-1.5 text-xs border-r border-slate-800 cursor-pointer select-none ${
                  activePath === tab.path
                    ? 'bg-slate-950 text-amber-300 border-t-2 border-t-amber-400 font-medium'
                    : 'text-slate-400 hover:bg-slate-900/60'
                }`}
              >
                <span>{tab.name}</span>
                {tab.isDirty && <span className="w-2 h-2 rounded-full bg-amber-400" title="Unsaved changes" />}
                <button
                  onClick={(e) => {
                    e.stopPropagation();
                    setActiveTabs((prev) => prev.filter((t) => t.path !== tab.path));
                    if (activePath === tab.path) setActivePath(null);
                  }}
                  className="hover:text-rose-400 p-0.5 rounded"
                >
                  <X className="w-3 h-3" />
                </button>
              </div>
            ))}
          </div>

          {/* MONACO / DIFF EDITOR CONTAINER */}
          <div className="flex-1 relative">
            {stagedDiff ? (
              <div className="h-full flex flex-col">
                <div className="flex items-center justify-between bg-amber-500/10 border-b border-amber-500/30 px-4 py-2 text-xs">
                  <span className="text-amber-300 font-semibold flex items-center gap-2">
                    <Sparkles className="w-4 h-4" /> Staged Shadow Diff ({stagedDiff.agentId.toUpperCase()})
                  </span>
                  <div className="flex items-center gap-2">
                    <button
                      onClick={acceptDiff}
                      className="flex items-center gap-1 px-3 py-1 bg-emerald-600 hover:bg-emerald-500 text-white rounded font-bold"
                    >
                      <Check className="w-3.5 h-3.5" /> Accept Staging
                    </button>
                    <button
                      onClick={rejectDiff}
                      className="flex items-center gap-1 px-3 py-1 bg-rose-600 hover:bg-rose-500 text-white rounded font-bold"
                    >
                      <X className="w-3.5 h-3.5" /> Reject
                    </button>
                  </div>
                </div>
                <div className="flex-1">
                  <DiffEditor
                    original={stagedDiff.originalCode}
                    modified={stagedDiff.stagedCode}
                    language={getLanguageFromPath(stagedDiff.path)}
                    theme="vs-dark"
                    options={{ readOnly: true, renderSideBySide: true }}
                  />
                </div>
              </div>
            ) : currentTab ? (
              <Editor
                height="100%"
                path={currentTab.path}
                language={currentTab.language}
                value={currentTab.content}
                theme="vs-dark"
                onMount={handleEditorMount}
                onChange={(val) => {
                  const updated = val || '';
                  setActiveTabs((prev) =>
                    prev.map((t) =>
                      t.path === currentTab.path
                        ? { ...t, content: updated, isDirty: updated !== t.originalContent }
                        : t
                    )
                  );
                }}
                options={{
                  fontSize: 13,
                  minimap: { enabled: true },
                  smoothScrolling: true,
                  fontFamily: 'Fira Code, JetBrains Mono, monospace'
                }}
              />
            ) : (
              <div className="flex flex-col items-center justify-center h-full text-slate-600">
                <Command className="w-12 h-12 mb-2 opacity-40" />
                <p className="text-sm">Select a file to open or press <kbd className="px-1.5 py-0.5 bg-slate-800 text-slate-300 rounded">Ctrl+K</kbd> for Inline Composer</p>
              </div>
            )}

            {/* INLINE COMPOSER FLOATING OVERLAY (Ctrl+K) */}
            {composerOpen && (
              <div
                className="absolute z-50 w-96 bg-slate-900 border border-amber-500/40 rounded-lg shadow-2xl p-3 flex flex-col gap-2"
                style={{ top: composerPosition.top, left: composerPosition.left }}
              >
                <div className="flex items-center justify-between border-b border-slate-800 pb-1.5">
                  <span className="text-xs font-bold text-amber-400 flex items-center gap-1.5">
                    <Sparkles className="w-3.5 h-3.5" /> Inline Agent Prompt ({selectedAgent.toUpperCase()})
                  </span>
                  <button onClick={() => setComposerOpen(false)} className="text-slate-400 hover:text-slate-200">
                    <X className="w-3.5 h-3.5" />
                  </button>
                </div>
                <textarea
                  autoFocus
                  value={composerPrompt}
                  onChange={(e) => setComposerPrompt(e.target.value)}
                  placeholder="Ask agent to generate, refactor, or fix code..."
                  className="w-full bg-slate-950 border border-slate-800 rounded p-2 text-xs text-slate-100 focus:outline-none focus:border-amber-400 resize-none h-20"
                />
                <div className="flex justify-end gap-2">
                  <button
                    onClick={handleComposerSubmit}
                    className="px-3 py-1 bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold rounded text-xs flex items-center gap-1"
                  >
                    <Play className="w-3 h-3 fill-current" /> Dispatch
                  </button>
                </div>
              </div>
            )}
          </div>
        </main>
      </div>

      {/* 4. TERMINAL DRAWER */}
      <div className={`relative z-10 border-t border-slate-800 bg-slate-950/90 flex flex-col transition-all duration-200 ${terminalOpen ? 'h-48' : 'h-8'}`}>
        <div className="flex items-center justify-between px-3 py-1.5 bg-slate-900 border-b border-slate-800/80 text-xs">
          <span className="font-semibold text-slate-400 flex items-center gap-2">
            <TermIcon className="w-3.5 h-3.5 text-emerald-400" /> Terminal / Exec Engine (`POST /api/ide/exec`)
          </span>
          <button 
            onClick={() => setTerminalOpen(!terminalOpen)}
            className="text-slate-400 hover:text-slate-200 text-xs"
          >
            {terminalOpen ? 'Collapse' : 'Expand'}
          </button>
        </div>

        {terminalOpen && (
          <div className="flex-1 flex flex-col p-2 overflow-hidden">
            <div className="flex-1 overflow-y-auto space-y-2 mb-2 pr-1 font-mono text-xs">
              {execHistory.map((item) => (
                <div key={item.id} className="border-l-2 border-slate-700 pl-2">
                  <div className="flex items-center gap-2 text-slate-500 text-[10px]">
                    <span>[{item.timestamp}]</span>
                    <span className="text-amber-400 font-bold">$ {item.command}</span>
                  </div>
                  <pre className="text-slate-300 whitespace-pre-wrap mt-0.5">{item.output}</pre>
                </div>
              ))}
            </div>
            
            <div className="flex items-center gap-2 bg-slate-900 px-2 py-1 rounded border border-slate-800">
              <span className="text-emerald-400 font-bold">$</span>
              <input
                type="text"
                value={currentCommand}
                onChange={(e) => setCurrentCommand(e.target.value)}
                onKeyDown={(e) => e.key === 'Enter' && executeCommand()}
                disabled={isExecuting}
                placeholder="Enter command..."
                className="flex-1 bg-transparent text-xs text-slate-100 focus:outline-none"
              />
              <button 
                onClick={executeCommand} 
                disabled={isExecuting}
                className="p-1 text-slate-400 hover:text-amber-400 disabled:opacity-50"
              >
                {isExecuting ? <RefreshCw className="w-3.5 h-3.5 animate-spin" /> : <Play className="w-3.5 h-3.5" />}
              </button>
            </div>
          </div>
        )}
      </div>
    </div>
  );
};

function renderTree(
  nodes: FileNode[], 
  onOpen: (path: string) => void, 
  onDelete: (path: string) => void,
  activePath: string | null,
  activeTabs: ActiveTab[]
) {
  return nodes.map((node) => {
    const isDirty = activeTabs.some((t) => t.path === node.path && t.isDirty);
    const isActive = activePath === node.path;

    if (node.type === 'directory') {
      return (
        <details key={node.path} className="group ml-1">
          <summary className="flex items-center gap-1 px-1.5 py-1 hover:bg-slate-800/60 rounded cursor-pointer text-slate-300 select-none">
            <ChevronRight className="w-3 h-3 group-open:rotate-90 transition-transform text-slate-500" />
            <Folder className="w-3.5 h-3.5 text-amber-500" />
            <span>{node.name}</span>
          </summary>
          <div className="pl-3 border-l border-slate-800 ml-2">
            {node.children && renderTree(node.children, onOpen, onDelete, activePath, activeTabs)}
          </div>
        </details>
      );
    }

    return (
      <div
        key={node.path}
        onClick={() => onOpen(node.path)}
        className={`flex items-center justify-between px-1.5 py-1 rounded cursor-pointer group ${
          isActive ? 'bg-slate-800 text-amber-300 font-medium' : 'hover:bg-slate-800/40 text-slate-400'
        }`}
      >
        <div className="flex items-center gap-1.5 truncate">
          <File className="w-3.5 h-3.5 text-slate-500 shrink-0" />
          <span className="truncate">{node.name}</span>
          {isDirty && <span className="w-1.5 h-1.5 rounded-full bg-amber-400 shrink-0" />}
        </div>
        <button
          onClick={(e) => {
            e.stopPropagation();
            onDelete(node.path);
          }}
          className="opacity-0 group-hover:opacity-100 hover:text-rose-400 p-0.5"
          title="Delete file"
        >
          <Trash2 className="w-3 h-3" />
        </button>
      </div>
    );
  });
}

function getLanguageFromPath(path: string): string {
  const ext = path.split('.').pop()?.toLowerCase();
  switch (ext) {
    case 'ts':
    case 'tsx': return 'typescript';
    case 'js':
    case 'jsx': return 'javascript';
    case 'json': return 'json';
    case 'html': return 'html';
    case 'css': return 'css';
    case 'md': return 'markdown';
    default: return 'plaintext';
  }
}

### 4. Conductor Backend Integration Specifications

To support complete execution of Forge Pro IDE, the backend conductor must fulfill the following endpoint contracts:

1. **`DELETE /api/ide/file`** (Required for file tree deletion):
   - **Body**: `{ "path": "relative/file/path.ext" }`
   - **Response**: `{ "success": true, "path": "..." }`

2. **`POST /api/omni/acp/agents/dispatch`** (Required for AI Worker Agent execution):
   - Accepts `AgentDispatchPayload` detailed above.
   - Evaluates code range or full file buffer, triggers model inference (Codex/Claude/Aider), returns staged code diff.
